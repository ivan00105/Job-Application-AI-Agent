% User Flow Diagram for Job Application System
% Requires: \usepackage{tikz}
% Fixed version with better positioning

\begin{figure}[h]
\centering
\begin{tikzpicture}[
    node distance = 1.2cm and 1.5cm,
    auto,
    % Define styles
    startstop/.style = {
        rectangle,
        rounded corners,
        minimum width = 2.5cm,
        minimum height = 0.8cm,
        text centered,
        draw = black,
        fill = blue!20
    },
    process/.style = {
        rectangle,
        minimum width = 2.5cm,
        minimum height = 0.8cm,
        text centered,
        draw = black,
        fill = yellow!20
    },
    decision/.style = {
        diamond,
        aspect = 2,
        minimum width = 2cm,
        minimum height = 1cm,
        text centered,
        draw = black,
        fill = green!20
    },
    io/.style = {
        trapezium,
        trapezium left angle = 70,
        trapezium right angle = 110,
        minimum width = 2.5cm,
        minimum height = 0.8cm,
        text centered,
        draw = black,
        fill = orange!20
    },
    arrow/.style = {
        ->,
        >=stealth
    }
]

    % Start
    \node [startstop] (start) {Start};
    
    % Upload CV
    \node [io, below of = start] (upload) {Upload CV};
    
    % Parse CV
    \node [process, below of = upload] (parse) {Parse CV\\Extract Profile};
    
    % Search Jobs
    \node [io, below of = parse] (search) {Search Jobs\\(Semantic)};
    
    % View Matches
    \node [process, below of = search] (matches) {View Job\\Matches};
    
    % Select Job Decision
    \node [decision, below of = matches] (select) {Select\\Job?};
    
    % Generate Tailored CV
    \node [process, below of = select] (generate) {Generate\\Tailored CV};
    
    % Review CV Decision
    \node [decision, below of = generate] (review) {Approve\\CV?};
    
    % Edit CV
    \node [process, left of = review, xshift = -2.5cm] (edit) {Edit CV};
    
    % Interview Prep Decision
    \node [decision, below of = review] (interview) {Interview\\Prep?};
    
    % Interview Prep
    \node [process, right of = interview, xshift = 2.5cm] (prep) {AI Interview\\Prep};
    
    % Apply
    \node [io, below of = interview] (apply) {Apply to Job};
    
    % End
    \node [startstop, below of = apply] (end) {End};
    
    % Arrows - Main flow
    \draw [arrow] (start) -- (upload);
    \draw [arrow] (upload) -- (parse);
    \draw [arrow] (parse) -- (search);
    \draw [arrow] (search) -- (matches);
    \draw [arrow] (matches) -- (select);
    \draw [arrow] (select) -- node [right] {Yes} (generate);
    \draw [arrow] (generate) -- (review);
    \draw [arrow] (review) -- node [right] {Yes} (interview);
    \draw [arrow] (interview) -- node [right] {No} (apply);
    \draw [arrow] (apply) -- (end);
    
    % Arrows - Loops and branches
    \draw [arrow] (select) -| node [near start, above] {No} (search);
    \draw [arrow] (review) -- node [above] {Edit} (edit);
    \draw [arrow] (edit) |- (review);
    \draw [arrow] (interview) -- node [above] {Yes} (prep);
    \draw [arrow] (prep) |- (apply);
    
\end{tikzpicture}
\caption{User Flow Diagram: Job Application Process}
\label{fig:user_flow}
\end{figure}
