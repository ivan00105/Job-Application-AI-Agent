\documentclass{article}

\usepackage{tikz}
\usetikzlibrary{positioning} % for "below=of ..." and "left=of ..."
\usetikzlibrary{arrows.meta} % for >=Stealth

\begin{document}

\begin{figure}[h]
\centering
\begin{tikzpicture}[
    node distance = 1cm and 1cm,
    auto,
    startstop/.style = {
        rectangle, rounded corners = 3pt,
        minimum width = 2.2cm, minimum height = 0.7cm,
        text centered, draw = black, fill = blue!20,
        font = \footnotesize
    },
    process/.style = {
        rectangle,
        minimum width = 2.2cm, minimum height = 0.7cm,
        text centered, draw = black, fill = yellow!20,
        font = \footnotesize
    },
    decision/.style = {
        diamond, aspect = 1.5,
        minimum width = 1.6cm, minimum height = 0.7cm,
        text centered, draw = black, fill = green!20,
        font = \footnotesize
    },
    io/.style = {
        rectangle, rounded corners = 3pt,
        minimum width = 2.2cm, minimum height = 0.7cm,
        text centered, draw = black, fill = orange!20,
        font = \footnotesize
    },
    arrow/.style = {->, >=Stealth}
]

    % Top row nodes
    \node[startstop] (start) {Start};
    \node[io,      right=of start]    (upload)   {Upload CV};
    \node[process, right=of upload]   (parse)    {Parse CV};
    \node[io,      right=of parse]    (search)   {Search Jobs};
    \node[process, right=of search]   (matches)  {View Matches};
    \node[decision,right=of matches]  (select)   {Select?};

    % Bottom row nodes
    \node[process, below=1.2cm of parse]   (generate) {Generate CV};
    \node[decision,right=of generate]      (review)   {Approve?};
    \node[process, below=1.2cm of select]  (edit)     {Edit CV};
    \node[decision,right=of review]        (interview){Interview Prep?};
    \node[process,right=of interview]      (prep)     {AI Prep};
    \node[io,      right=of prep]          (apply)    {Apply};
    \node[startstop, right=of apply]       (end)      {End};

    % Connections top row
    \draw[arrow] (start) -- (upload);
    \draw[arrow] (upload) -- (parse);
    \draw[arrow] (parse) -- (search);
    \draw[arrow] (search) -- (matches);
    \draw[arrow] (matches) -- (select);
    \draw[arrow] (select) |- node[near start, left, font=\tiny] {No} (search);

    % Connections to bottom row
    \draw[arrow] (select) |- node[left, font=\tiny] {Yes} (generate);
    \draw[arrow] (generate) -- (review);
    \draw[arrow] (review) -- node[above, font=\tiny] {Yes} (interview);
    \draw[arrow] (review) -- node[right, font=\tiny] {No} (edit);
    \draw[arrow] (edit) |- (generate);
    \draw[arrow] (interview) -- node[above, font=\tiny] {No} (apply);
    \draw[arrow] (interview) -- node[above, font=\tiny] {Yes} (prep);
    \draw[arrow] (prep) -- (apply);
    \draw[arrow] (apply) -- (end);
    
\end{tikzpicture}
\caption{User Flow Diagram: Job Application Process}
\label{fig:user_flow}
\end{figure}
